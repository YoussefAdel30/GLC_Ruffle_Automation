python - <<'PY'
p='/FMS/ERM801/erm/config/erm_config.json'
lines=open(p).read().splitlines(True)
for i,ln in enumerate(lines):
    if '"name"' in ln and 'vertica.enabled' in ln:
        print('FOUND', i+1, ln.rstrip())
        print(' NEXT', lines[i+1].rstrip())
        if '"value"' in lines[i+1]:
            indent=lines[i+1][:len(lines[i+1])-len(lines[i+1].lstrip())]
            comma=',' if lines[i+1].rstrip().endswith(',') else ''
            lines[i+1]='%s"value" : "true"%s\n' % (indent, comma)
            print(' NEW', lines[i+1].rstrip())
        break
open(p,'w').writelines(lines)
PY
