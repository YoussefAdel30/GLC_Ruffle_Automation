SELECT column_name, data_type
FROM all_tab_columns
WHERE owner = 'INTERNAL_DFMS' AND table_name = 'QB_CUSTOM_QUERIES'
ORDER BY column_id;

SELECT COUNT(*) AS queries FROM INTERNAL_DFMS.QB_CUSTOM_QUERIES;
SELECT COUNT(*) AS folders FROM INTERNAL_DFMS.QB_QUERIES_FOLDERS;
SELECT COUNT(*) AS seq_items FROM INTERNAL_DFMS.QB_QSEQUENCE_ITEMS;
SELECT COUNT(*) AS custom_tables FROM INTERNAL_DFMS.QB_CUSTOM_TABLES;


SELECT owner, table_name
FROM all_tables
WHERE owner IN ('FMSUSER','INTERNAL_DFMS')
  AND REGEXP_LIKE(table_name, 'TOOLBAR|DASHBOARD|WIDGET|QB_')
ORDER BY 1,2;
